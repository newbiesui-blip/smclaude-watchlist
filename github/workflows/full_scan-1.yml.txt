name: Full Market Scan

on:
  schedule:
    # Every 4 hours, on the hour. Best-effort timing like the 15-min job --
    # can run a few minutes late under GitHub's load, that's normal.
    - cron: "0 */4 * * *"
  workflow_dispatch: {}   # manual trigger from the Actions tab, for testing

permissions:
  contents: write

jobs:
  full-scan:
    runs-on: ubuntu-latest
    # A full 150-symbol x 4-timeframe scan does a lot of API calls with a
    # polite delay between each -- give it real headroom instead of the
    # default 6h cap being the only thing stopping a runaway job.
    timeout-minutes: 30
    steps:
      - name: Checkout repo
        uses: actions/checkout@v4

      - name: Set up Python
        uses: actions/setup-python@v5
        with:
          python-version: "3.11"

      - name: Install dependencies
        run: pip install pandas numpy requests

      - name: Run full scan
        env:
          TG_BOT_TOKEN: ${{ secrets.TG_BOT_TOKEN }}
          TG_CHAT_ID: ${{ secrets.TG_CHAT_ID }}
        run: python full_scan.py

      - name: Commit updated watchlist (if changed)
        run: |
          git config user.name "watchlist-bot"
          git config user.email "actions@users.noreply.github.com"
          if ! git diff --quiet smc_watchlist.json 2>/dev/null; then
            git add smc_watchlist.json
            git commit -m "Auto full-scan: add qualifying setups [skip ci]"
            git push
          else
            echo "No changes to commit."
          fi
